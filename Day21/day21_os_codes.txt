#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>

int main() {
    char name[50];
    char regno[20];
    int age;

    // Parent process: Read user input
    printf("Enter name: ");
    scanf("%s", name);
    printf("Enter registration number: ");
    scanf("%s", regno);
    printf("Enter age: ");
    scanf("%d", &age);

    pid_t pid = fork(); // Create a child process

    if (pid < 0) {
        printf("Fork failed\n");
        return 1;
    } else if (pid == 0) {
        // Child process: Print the output
        printf("\n--- Child Process Output ---\n");
        printf("Name: %s\n", name);
        printf("Registration Number: %s\n", regno);
        printf("Age: %d\n", age);
    } else {
        // Parent process: Wait for the child to finish
        wait(NULL);
    }

    return 0;
}

::: using pipe
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>

int main() {
    char name[50];
    char regno[20];
    int age;
    int pipefd[2]; // Pipe file descriptors
    pid_t pid;

    // Create pipe
    if (pipe(pipefd) == -1) {
        perror("Pipe failed");
        return 1;
    }

    // Fork a child process
    pid = fork();

    if (pid < 0) {
        perror("Fork failed");
        return 1;
    }

    if (pid > 0) {
        // Parent Process: Read input and send data through the pipe
        printf("Enter name: ");
        scanf("%s", name);
        printf("Enter registration number: ");
        scanf("%s", regno);
        printf("Enter age: ");
        scanf("%d", &age);

        close(pipefd[0]); // Close unused read end
        write(pipefd[1], name, sizeof(name));
        write(pipefd[1], regno, sizeof(regno));
        write(pipefd[1], &age, sizeof(age));
        close(pipefd[1]); // Close write end after sending data

        // Wait for the child process to finish
        wait(NULL);
    } else {
        // Child Process: Read data from the pipe and display output
        close(pipefd[1]); // Close unused write end
        read(pipefd[0], name, sizeof(name));
        read(pipefd[0], regno, sizeof(regno));
        read(pipefd[0], &age, sizeof(age));
        close(pipefd[0]); // Close read end after receiving data

        printf("\n--- Child Process Output ---\n");
        printf("Name: %s\n", name);
        printf("Registration Number: %s\n", regno);
        printf("Age: %d\n", age);
    }

    return 0;
}
