#include <stdio.h>
int main(){

int n;
long long int fact =1;
printf("Enter Number for find factorial: ");
scanf("%d",&n);


if (n<0){
    printf("Type postive number! \n");
}

    for(int i=1;i<=n;i++)
    {
		fact*=i;
    }
	printf("Ans: ",fact);

	return 0;

}

#include <stdio.h>
int main(){

	int a=0, b=1,next;

	int n;
	printf("Enter a number: ");
	scanf("%d",&n);

	for(int i=1;i<=n;i++)
	{
		 if(i == 1)
		 {
			 printf("%d",a);
		 }else if(i == 2){
			 printf("%d",b);
		 }else{
			 next = a+b;
			 printf("%d",next);
			 a=b;
			 b=next;
		 }
		 printf("\n");
	}

	return 0;
}
